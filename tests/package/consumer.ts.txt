import { loadTexture, loadKTX2 } from "@adui/three/textures";
import { captureFrame } from "@adui/three/capture";
import type { CaptureRegion } from "@adui/three/capture";
import { createInstancedMesh, computeObjectBounds } from "@adui/three/geometry";
import { loadHDR, loadEXR, attachEnvironment } from "@adui/three/environment";
import { ResourceScope, ThreeApp, OrthographicThreeApp } from "@adui/three";
import type { AnyThreeApp } from "@adui/three";
import { attachOrbitControls, attachTransformControls } from "@adui/three/controls";
import { cloneGLTF, loadGLTF } from "@adui/three/loaders";
import type { GLTFInstance, GLTFModel, LoadGLTFOptions } from "@adui/three/loaders";
import { attachPicking } from "@adui/three/interaction";
import { attachDOMLabels } from "@adui/three/labels";
import { attachGLTFAnimations, followPath } from "@adui/three/animation";
import { attachBloom, attachPostprocessing } from "@adui/three/postprocessing";
import { computeObjectFrame, fitObject, transitionToObject, computeOrthographicFrame, fitOrthographicObject, transitionOrthographicToObject, captureCameraView, restoreCameraView } from "@adui/three/camera";
import { transitionToCameraView } from "@adui/three/camera";
import type { CameraTransitionResult, CameraViewTransitionResult } from "@adui/three/camera";
import { Box3, BoxGeometry, Color, LineCurve3, Matrix4, Mesh, MeshBasicMaterial, Vector3 } from "three";

// Compile only: all library values come from the installed archive, without workspace paths.
export async function consume(container: HTMLElement, overlay: HTMLElement, model: GLTFModel) {
  const app = new ThreeApp(container, { maxPixelRatio: 2 });
  const bounds: Box3 = computeObjectBounds(model.scene);
  const measuredSize: Vector3 = bounds.getSize(new Vector3());
  void measuredSize;
  const exr = await loadEXR(app, "/environment.exr", { signal: app.signal });
  attachEnvironment(app, exr, { background: true });
  exr.dispose();
  const orthographic = new OrthographicThreeApp(container, { camera: { frustumHeight: 8, near: 0 } });
  const drewPerspective: boolean = app.render(), drewOrthographic: boolean = orthographic.render();
  void [drewPerspective, drewOrthographic];
  const span: number = orthographic.camera.top - orthographic.camera.bottom;
  const savedView = captureCameraView(orthographic);
  const restoredView = restoreCameraView(orthographic, savedView);
  const viewMotion = transitionToCameraView(orthographic, savedView, { duration: 1, signal: orthographic.signal });
  const viewResult: CameraViewTransitionResult = await viewMotion.finished;
  if (viewResult.status === "completed" && viewResult.view.kind === "orthographic") {
    const height: number = viewResult.view.top - viewResult.view.bottom; void height;
  }
  viewMotion.cancel(); viewMotion.dispose();
  viewMotion.pause(); viewMotion.resume();
  if (viewMotion.state === "paused") viewMotion.resume();
  if (restoredView.kind === "orthographic") { const center: number = restoredView.centerX; void center; }
  const orthographicFrame = computeOrthographicFrame(orthographic.camera, model.scene);
  const fittedZoom: number = fitOrthographicObject(orthographic, model.scene).zoom;
  void [orthographicFrame, fittedZoom];
  const orthographicMotion = transitionOrthographicToObject(orthographic, model.scene);
  const orthographicResult = await orthographicMotion.finished;
  if (orthographicResult.status === "completed") { const zoom: number = orthographicResult.frame.zoom; void zoom; }
  orthographicMotion.cancel();
  orthographicMotion.pause(); orthographicMotion.resume();
  const aspect: number = app.camera.aspect;
  const shared: AnyThreeApp = orthographic;
  attachOrbitControls(shared);
  attachDOMLabels(orthographic, overlay);
  attachPicking(orthographic, { objects: () => [] });
  attachBloom(orthographic);
  loadGLTF(orthographic, "/model.glb");
  loadTexture(orthographic, "/image.png");
  captureFrame(orthographic);
  attachGLTFAnimations(orthographic, model);
  cloneGLTF(orthographic, model);
  void [span, aspect];
  // @ts-expect-error Framing helpers currently require a perspective host.
  fitObject(orthographic, model.scene);
  orthographic.dispose();
  const region: CaptureRegion = { x: 0, y: 0, width: 1, height: 1 };
  const image: Blob = await captureFrame(app, { type: "image/webp", quality: 0.8, signal: app.signal, region });
  void image;
  const contextLost: boolean = app.contextLost;
  void contextLost;
  const scope = new ResourceScope();
  const object = new Mesh(scope.track(new BoxGeometry()), scope.track(new MeshBasicMaterial()));
  app.scene.add(object);
  const instances = createInstancedMesh(app, { geometry: object.geometry, material: object.material, capacity: 10 });
  instances.setMatrixAt(0, new Matrix4().makeTranslation(1, 0, 0));
  instances.setColorAt(0, new Color(0xff0000));
  instances.setCount(1);
  instances.dispose();
  const { controls } = attachOrbitControls(app);
  controls.target.set(0, 1, 0);
  const gizmo = attachTransformControls(app, { object, orbitControls: controls, mode: "translate", space: "world", size: 1 });
  gizmo.controls.setMode("rotate");
  gizmo.detach();
  gizmo.attach(object);
  gizmo.dispose();
  const options: LoadGLTFOptions = { signal: app.signal, addToScene: false };
  const loaded: GLTFModel = await loadGLTF(app, "/sample.glb", options);
  const instance: GLTFInstance = cloneGLTF(app, loaded, { addToScene: false });
  const clonedAnimations = attachGLTFAnimations(app, instance);
  if (clonedAnimations.clips.length) clonedAnimations.play(0);
  instance.resources.track({ dispose: () => console.log("实例结束") });
  instance.dispose();
  const picking = attachPicking(app, { objects: () => [loaded.scene], updateSkinnedBounds: true });
  picking.on("click", ({ intersection }) => {
    intersection?.object.position.set(0, 0, 0);
  });
  const labels = attachDOMLabels(app, overlay);
  labels.add({ object: loaded.scene, element: document.createElement("span") });
  const animations = attachGLTFAnimations(app, model);
  animations.setPlaybackRate(0.5);
  const playbackRate: number = animations.playbackRate; void playbackRate;
  const offLoop = animations.on("loop", (event) => {
    const traversals: number = event.loopDelta;
    console.log(event.clip.name, traversals);
  });
  const offFinished = animations.on("finished", (event) => {
    const direction: number = event.direction;
    console.log(event.action.getClip().name, direction);
  });
  offLoop(); offFinished();
  animations.play(0, { loop: "once" });
  const path = followPath(app, loaded.scene, {
    curve: new LineCurve3(new Vector3(), new Vector3(2, 0, 0)), duration: 2,
    loop: "pingpong", repetitions: 2, orientToPath: true, scope: loaded.resources, signal: app.signal,
  });
  path.setPlaybackRate(0.5);
  const pathRate: number = path.playbackRate; void pathRate;
  path.seek(0.5); path.pause(); path.resume(); path.cancel();
  const pathResult = await path.finished;
  if (pathResult.status === "failed") console.error(pathResult.error);
  animations.play(0, { loop: "pingpong", repetitions: 2 });
  animations.seek(0);
  animations.resume();
  animations.crossFadeTo(0, { duration: 0.3, loop: "repeat", repetitions: 3 });
  const blendProgress: number = animations.blendProgress;
  const blending: boolean = animations.blending;
  void blendProgress; void blending;
  computeObjectFrame(app.camera, object, { direction: [1, 1, 1] });
  fitObject(app, object, { controls });
  const motion = transitionToObject(app, object, { controls, duration: 1, signal: app.signal });
  const result: CameraTransitionResult = await motion.finished;
  if (result.status === "completed") result.frame.position.toArray();
  if (result.status === "failed") console.error(result.error);
  app.onUpdate(({ delta, elapsed }) => {
    object.rotation.y += delta;
    object.position.y = Math.sin(elapsed);
  });
  app.onAfterRender(() => labels.update());
  app.start();
  app.stop();
  app.render();
  const detachPipeline = app.setRenderPipeline({
    render: ({ delta }) => { object.rotation.y += delta; app.renderer.render(app.scene, app.camera); },
    resize: ({ width, height, pixelRatio }) => { console.log(width, height, pixelRatio); },
  });
  detachPipeline();
  const bloom = attachBloom(app, { strength: 1, radius: 0.4, threshold: 0.9 });
  bloom.pass.enabled = false;
  bloom.dispose();
  const effects = attachPostprocessing(app, { bloom: {}, outline: { selectedObjects: [object], visibleEdgeColor: 0xff8800 } });
  effects.outline!.selectedObjects = [loaded.scene];
  effects.dispose();
  const environment = await loadHDR(app, "/studio.hdr", { signal: app.signal, onProgress: ({ ratio }) => { const value: number | null = ratio; void value; } });
  const environmentBinding = attachEnvironment(app, environment, { background: true, intensity: 1.5 });
  environmentBinding.dispose();
  environment.dispose();
  const textureAsset = await loadTexture(app, "/image.png", { colorSpace: "srgb", flipY: true, signal: app.signal });
  textureAsset.texture.repeat.set(2, 2);
  textureAsset.resources.track({ dispose: () => { object.material.map = null; } });
  object.material.map = textureAsset.texture;
  textureAsset.dispose();
  const compressed = await loadKTX2(app, "/map.ktx2", { transcoderPath: { js: "/basis.js", wasm: "/basis.wasm" }, workerLimit: 1, signal: app.signal });
  object.material.map = compressed.texture;
  compressed.resources.track({ dispose: () => { object.material.map = null; } });
  compressed.dispose();
  app.dispose();
  scope.dispose();
}

export function invalidInputChecks(app: ThreeApp, object: Mesh) {
  // @ts-expect-error duration must remain numeric in the published declaration
  transitionToObject(app, object, { duration: "1" });
  // @ts-expect-error picking requires a dynamic objects getter
  attachPicking(app, {});
}
